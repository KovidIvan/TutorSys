# Быстрый запуск существующего проекта TutorSys в PowerShell

# Перейти в корень репозитория и создать окружение
cd D:\TutorSys
python -m venv .venv
.venv\Scripts\Activate.ps1

# Установить зависимости и создать локальный файл конфигурации
python -m pip install -r requirements.txt
Copy-Item .env.example .env

# Задать уникальный SECRET_KEY в .env.
# Если DB_ENGINE и DB_NAME пустые, используется локальная SQLite.

# Для PostgreSQL укажите в .env непустые DB_ENGINE и DB_NAME, например:
# DB_ENGINE=django.db.backends.postgresql
# DB_NAME=nagruzka
# DB_USER=postgres
# DB_PASSWORD=ваш-пароль
# DB_HOST=localhost
# DB_PORT=5432
# Настройки, маршруты и шаблон входа уже включены в проект.

# Применить миграции и загрузить данные для демо
python manage.py migrate
python manage.py shell -c "from core.seed import run; run()"

# Проверить проект и запустить тесты
python manage.py check
python manage.py test core -v 2

# Запустить сервер и открыть приложение
python manage.py runserver
# http://127.0.0.1:8000/accounts/login/
